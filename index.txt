-- 30. Update salary then rollback

-- 31. Create savepoint and rollback to it
START TRANSACTION;

SAVEPOINT before_update;

UPDATE Employee_Details SET salary = salary + 5000;

ROLLBACK TO before_update;

COMMIT;

-- 32. Multiple inserts with savepoint
START TRANSACTION;

INSERT INTO Employee_Details (employee_name, dept_name, salary)

SAVEPOINT after_first_insert;

INSERT INTO Employee_Details (employee_name, dept_name, salary)
VALUES ('Arjun', 'HR', 45000);

ROLLBACK TO after_first_insert;

COMMIT;

-- 33. Delete HR employees then rollback

-- 34. Complex transaction with savepoint
START TRANSACTION;
